Public Sub UI_ApplySheets()
    Dim ws As Worksheet, lastCol As Long, used As Range
    On Error Resume Next
    For Each ws In ThisWorkbook.Worksheets
        If Left$(ws.Name, 5) = "zz_Ex" Or ws.Name = "zz_PaletteCheck" Then
            Set used = ws.UsedRange
            used.Font.Name = "Meiryo UI": used.Font.Size = 10
            lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
            With ws.Range(ws.Cells(1, 1), ws.Cells(1, lastCol))
                .Interior.Color = UI_LIGHT: .Font.Color = UI_NAVY
                .Font.Bold = True: .RowHeight = 24
            End With
            used.Borders.Color = RGB(230, 230, 230)
        End If
    Next
    On Error GoTo 0
End Sub

Private Sub BaseForm(ByVal frm As Object, ByVal cap As String, ByVal w As Single, ByVal h As Single)
    On Error Resume Next
    frm.Caption = cap: frm.Width = w: frm.Height = h
    frm.BackColor = UI_FORM: frm.Font.Name = "Meiryo UI": frm.Font.Size = 9
    On Error GoTo 0
End Sub

Private Sub Rect(ByVal box As Object, ByVal n As String, ByVal l As Single, _
    ByVal t As Single, ByVal w As Single, ByVal h As Single)
    Dim c As Object
    On Error Resume Next
    Set c = box.Controls(n)
    c.Left = l: c.Top = t: c.Width = w: c.Height = h
    c.Font.Name = "Meiryo UI": c.Font.Size = 9
    On Error GoTo 0
End Sub

Private Sub CaptionOf(ByVal box As Object, ByVal n As String, ByVal text As String)
    On Error Resume Next: box.Controls(n).Caption = text: On Error GoTo 0
End Sub

Private Sub HeaderBand(ByVal box As Object, ByVal n As String)
    On Error Resume Next: box.Controls(n).BackStyle = 1: box.Controls(n).BackColor = UI_LIGHT: On Error GoTo 0
End Sub

Private Sub LineOf(ByVal box As Object, ByVal n As String)
    On Error Resume Next: box.Controls(n).BackStyle = 1: box.Controls(n).BackColor = UI_NAVY: On Error GoTo 0
End Sub

Private Sub TitleOf(ByVal box As Object, ByVal n As String, ByVal size As Single)
    On Error Resume Next
    With box.Controls(n): .BackStyle = 0: .ForeColor = UI_NAVY: .Font.Name = "Meiryo UI": .Font.Size = size: .Font.Bold = True: End With
    On Error GoTo 0
End Sub

Private Sub SubtitleOf(ByVal box As Object, ByVal n As String)
    On Error Resume Next
    With box.Controls(n): .BackStyle = 0: .ForeColor = RGB(105, 105, 105): .Font.Name = "Meiryo UI": .Font.Size = 8.5: End With
    On Error GoTo 0
End Sub

Private Sub ToolName(ByVal box As Object, ByVal n As String)
    On Error Resume Next
    With box.Controls(n): .BackStyle = 0: .ForeColor = UI_NAVY: .Font.Bold = True: .Font.Name = "Meiryo UI": End With
    On Error GoTo 0
End Sub

Private Sub ToolDesc(ByVal box As Object, ByVal n As String)
    On Error Resume Next
    With box.Controls(n): .BackStyle = 0: .ForeColor = UI_TEXT: .Font.Size = 8: .WordWrap = True: End With
    On Error GoTo 0
End Sub

Private Sub ButtonOf(ByVal box As Object, ByVal n As String, ByVal primary As Boolean)
    On Error Resume Next
    With box.Controls(n)
        .BackColor = IIf(primary, RGB(230, 230, 230), UI_WHITE)
        .ForeColor = IIf(primary, UI_NAVY, UI_TEXT)
        .Font.Name = "Meiryo UI": .Font.Size = 9: .Font.Bold = primary
    End With
    On Error GoTo 0
End Sub

Private Sub TextBoxOf(ByVal box As Object, ByVal n As String, ByVal multi As Boolean)
    On Error Resume Next
    With box.Controls(n)
        .BackColor = UI_WHITE: .ForeColor = UI_TEXT: .BorderStyle = 1
        .MultiLine = multi: .WordWrap = multi
        If multi Then .ScrollBars = 2
    End With
    On Error GoTo 0
End Sub

Private Sub ComboOf(ByVal box As Object, ByVal n As String)
    On Error Resume Next: box.Controls(n).BackColor = UI_WHITE: box.Controls(n).ForeColor = UI_TEXT: On Error GoTo 0
End Sub

Private Sub CandidateOf(ByVal box As Object, ByVal n As String)
    On Error Resume Next
    With box.Controls(n)
        .BackStyle = 1: .BackColor = UI_WHITE: .ForeColor = UI_TEXT
        .BorderStyle = 1: .BorderColor = UI_LINE: .WordWrap = True
    End With
    On Error GoTo 0
End Sub
